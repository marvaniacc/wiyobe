   147→function exportCSV(filename: string, headers: string[], rows: Array<Array<string | number | null | undefined>>) {
   148→  const escape = (val: string | number | null | undefined) => {
   149→    const s = String(val ?? '')
   150→    if (/[",\n\r]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   151→    return s
   152→  }
   153→  const csv = [headers.map(escape).join(','), ...rows.map((r) => r.map(escape).join(','))].join('\n')
   154→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   155→  const url = URL.createObjectURL(blob)
   156→  const a = document.createElement('a')
   157→  a.href = url
   158→  a.download = filename
   159→  document.body.appendChild(a)
   160→  a.click()
   161→  document.body.removeChild(a)
   162→  URL.revokeObjectURL(url)
   163→}

   237→function getProviderProfile(u: AdminUser) {
   238→  if (u.doctor) return { type: 'DOCTOR' as ProviderType, label: u.doctor.specialty, verified: u.doctor.verified, city: u.doctor.city, country: u.doctor.country, name: u.name }
   239→  if (u.hospital) return { type: 'HOSPITAL' as ProviderType, label: u.hospital.name, verified: u.hospital.verified, city: u.hospital.city, country: u.hospital.country, name: u.hospital.name }
   240→  if (u.hotel) return { type: 'HOTEL' as ProviderType, label: u.hotel.name, verified: u.hotel.verified, city: u.hotel.city, country: u.hotel.country, name: u.hotel.name }
   241→  if (u.translator) return { type: 'TRANSLATOR' as ProviderType, label: u.translator.specialization, verified: u.translator.verified, city: u.translator.city, country: u.translator.country, name: u.name }
   242→  return null
   243→}

   856→function CommissionSection() {

   859→  const [rates, setRates] = React.useState<Record<ProviderType, string>>({ DOCTOR: '', HOSPITAL: '', HOTEL: '', TRANSLATOR: '' })
   860→  const [saving, setSaving] = React.useState(false)
   861→
   862→  React.useEffect(() => {
   863→    if (data?.rates) {
   864→      const map: Record<ProviderType, string> = { DOCTOR: '', HOSPITAL: '', HOTEL: '', TRANSLATOR: '' }
   865→      data.rates.forEach((r) => { map[r.providerType] = r.rate })
   866→      setRates(map)
   867→    }
   868→  }, [data])
   869→
   870→  async function save() {
   871→    setSaving(true)
   872→    try {
   873→      const payload = { rates: (Object.keys(rates) as ProviderType[]).map((pt) => ({ providerType: pt, rate: rates[pt] || '0' })) }
   874→      await apiPut('/api/admin/commission', payload)
   875→      toast.success(t('admin.commissionUpdated'))
   876→      refetch()
   877→    } catch (e: any) {
   878→      toast.error(e.message || t('admin.error'))
   879→    } finally {
   880→      setSaving(false)
   881→    }
   882→  }
   883→
   884→  if (loading) {
   885→    return (
   886→      <div>
   887→        <PageHeader title={t('admin.commissionTitle')} icon="percent" />
   888→        <LoadingCard lines={4} />
   889→      </div>
   890→    )
   891→  }
   892→  if (error || !data) return <ErrorState message={error || t('admin.error')} onRetry={refetch} />
   893→
   894→  const types: ProviderType[] = ['DOCTOR', 'HOSPITAL', 'HOTEL', 'TRANSLATOR']
   895→  const iconFor = (pt: ProviderType) => pt === 'DOCTOR' ? 'medical_services' : pt === 'HOSPITAL' ? 'local_hospital' : pt === 'HOTEL' ? 'hotel' : 'translate'
   896→
   897→  return (
   898→    <div className="animate-fade-in">
   899→      <PageHeader title={t('admin.commissionTitle')} description={t('admin.commissionDesc')} icon="percent" />
   900→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   901→        <Card className="lg:col-span-2">
   902→          <CardHeader>
   903→            <CardTitle className="text-base">{t('admin.commissionTitle')}</CardTitle>
   904→            <CardDescription>{t('admin.commissionDesc')}</CardDescription>
   905→          </CardHeader>
   906→          <CardContent className="flex flex-col gap-3">
   907→            {types.map((pt) => (
   908→              <div key={pt} className="flex flex-col gap-3 rounded-[14px] border border-divider bg-surface-secondary/40 p-4 sm:flex-row sm:items-center sm:justify-between">
   909→                <div className="flex items-center gap-3">
   910→                  <div className="flex size-10 items-center justify-center rounded-[10px] bg-primary/10 text-primary">
   911→                    <Icon name={iconFor(pt)} size={22} fill />
   912→                  </div>
   913→                  <div>
   914→                    <p className="text-sm font-medium text-foreground">{t(PROVIDER_TYPE_LABEL_KEY[pt])}</p>
   915→                    <p className="text-xs text-muted-foreground">{t('admin.currentRate')}: {rates[pt] || '0'}%</p>
   916→                  </div>
   917→                </div>
   918→                <div className="flex items-center gap-2">
   919→                  <Input
   920→                    type="number"
   921→                    min="0"
   922→                    max="100"
   923→                    step="0.5"
   924→                    value={rates[pt]}
   925→                    onChange={(e) => setRates((s) => ({ ...s, [pt]: e.target.value }))}
   926→                    className="h-10 w-32 text-end"
   927→                  />
   928→                  <span className="text-sm font-medium text-muted-foreground">%</span>

   933→              <Button onClick={save} disabled={saving} className="gap-1.5">

   935→              </Button>
   936→            </div>
   937→          </CardContent>
   938→        </Card>
   939→
   940→        <Card>
   941→          <CardHeader>
   942→            <CardTitle className="text-base">{t('admin.currentRate')}</CardTitle>
   943→          </CardHeader>
   944→          <CardContent className="flex flex-col gap-3">

  1561→function ReportsSection() {
  1562→  const { t, locale } = useT()

  1564→
  1565→  const report = React.useMemo(() => {
  1566→    if (!data?.entries) return null
  1567→    let commission = 0, commissionReversal = 0, patientCharge = 0, refundPatient = 0, payout = 0
  1568→    const byTypeMap: Record<string, number> = {}
  1569→    const dailyMap: Record<string, number> = {}
  1570→
  1571→    data.entries.forEach((e) => {
  1572→      const amt = parseFloat(e.amount)
  1573→      if (e.type === 'COMMISSION') {
  1574→        commission += amt
  1575→        const day = new Date(e.createdAt).toISOString().slice(0, 10)
  1576→        dailyMap[day] = (dailyMap[day] || 0) + amt
  1577→      } else if (e.type === 'REFUND_COMMISSION_REVERSAL') {
  1578→        commissionReversal += Math.abs(amt)
  1579→        const day = new Date(e.createdAt).toISOString().slice(0, 10)
  1580→        dailyMap[day] = (dailyMap[day] || 0) - Math.abs(amt)
  1581→      } else if (e.type === 'PATIENT_CHARGE') {
  1582→        patientCharge += amt
  1583→      } else if (e.type === 'REFUND_PATIENT') {
  1584→        refundPatient += Math.abs(amt)
  1585→      } else if (e.type === 'PAYOUT') {
  1586→        payout += Math.abs(amt)
  1587→      }
  1588→    })
  1589→
  1590→    const platformRevenue = commission - commissionReversal
  1591→    const dailyRevenue = Object.entries(dailyMap)
  1592→      .map(([date, amount]) => ({ date, amount }))
  1593→      .sort((a, b) => a.date.localeCompare(b.date))
  1594→      .slice(-14)
  1595→
  1596→    return { platformRevenue, commission, commissionReversal, patientCharge, refundPatient, payout, dailyRevenue }
  1597→  }, [data])
  1598→

  1616→
