   150→   150→function exportCSV(filename: string, headers: string[], rows: Array<Array<string | number | null | undefined>>) {
   151→   151→  const escape = (val: string | number | null | undefined) => {
   152→   152→    const s = String(val ?? '')
   153→   153→    if (/[",\n\r]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   154→   154→    return s
   155→   155→  }
   156→   156→  const csv = [headers.map(escape).join(','), ...rows.map((r) => r.map(escape).join(','))].join('\n')
   157→   157→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   158→   158→  const url = URL.createObjectURL(blob)
   159→   159→  const a = document.createElement('a')
   160→   160→  a.href = url
   161→   161→  a.download = filename
   162→   162→  document.body.appendChild(a)
   163→   163→  a.click()
   164→   164→  document.body.removeChild(a)
   165→   165→  URL.revokeObjectURL(url)
   166→   166→}

   509→   509→function useUserAction(onSuccess?: () => void) {
   510→   510→  const { t } = useT()
   511→   511→  const [busyId, setBusyId] = React.useState<string | null>(null)
   512→   512→  const run = React.useCallback(async (userId: string, action: 'approve' | 'suspend' | 'activate' | 'reject') => {
   513→   513→    setBusyId(userId)
   514→   514→    try {
   515→   515→      await apiPost('/api/admin/users', { userId, action })
   516→   516→      toast.success(t('admin.userUpdated'))
   517→   517→      onSuccess?.()
   518→   518→    } catch (e: any) {
   519→   519→      toast.error(e.message || t('admin.error'))
   520→   520→    } finally {
   521→   521→      setBusyId(null)
   522→   522→    }
   523→   523→  }, [onSuccess, t])
   524→   524→  return { busyId, run }
   525→   525→}
