   150→function exportCSV(filename: string, headers: string[], rows: Array<Array<string | number | null | undefined>>) {
   151→  const escape = (val: string | number | null | undefined) => {
   152→    const s = String(val ?? '')
   153→    if (/[",\n\r]/.test(s)) return `"${s.replace(/"/g, '""')}"`
   154→    return s
   155→  }
   156→  const csv = [headers.map(escape).join(','), ...rows.map((r) => r.map(escape).join(','))].join('\n')
   157→  const blob = new Blob(['\uFEFF' + csv], { type: 'text/csv;charset=utf-8;' })
   158→  const url = URL.createObjectURL(blob)
   159→  const a = document.createElement('a')
   160→  a.href = url
   161→  a.download = filename
   162→  document.body.appendChild(a)
   163→  a.click()
   164→  document.body.removeChild(a)
   165→  URL.revokeObjectURL(url)
   166→}

   240→function getProviderProfile(u: AdminUser) {
   241→  if (u.doctor) return { type: 'DOCTOR' as ProviderType, label: u.doctor.specialty, verified: u.doctor.verified, city: u.doctor.city, country: u.doctor.country, name: u.name }
   242→  if (u.hospital) return { type: 'HOSPITAL' as ProviderType, label: u.hospital.name, verified: u.hospital.verified, city: u.hospital.city, country: u.hospital.country, name: u.hospital.name }
   243→  if (u.hotel) return { type: 'HOTEL' as ProviderType, label: u.hotel.name, verified: u.hotel.verified, city: u.hotel.city, country: u.hotel.country, name: u.hotel.name }
   244→  if (u.translator) return { type: 'TRANSLATOR' as ProviderType, label: u.translator.specialization, verified: u.translator.verified, city: u.translator.city, country: u.translator.country, name: u.name }
   245→  return null
   246→}

   510→  const { t } = useT()
   511→  const [busyId, setBusyId] = React.useState<string | null>(null)

   873→function CommissionSection() {
   874→  const { t, locale } = useT()
   875→  const { data, loading, error, refetch } = useApi<{ rates: CommissionRate[] }>('/api/admin/commission')
   876→  const [rates, setRates] = React.useState<Record<ProviderType, string>>({ DOCTOR: '', HOSPITAL: '', HOTEL: '', TRANSLATOR: '' })
   877→  const [saving, setSaving] = React.useState(false)
   878→
   879→  React.useEffect(() => {
   880→    if (data?.rates) {
   881→      const map: Record<ProviderType, string> = { DOCTOR: '', HOSPITAL: '', HOTEL: '', TRANSLATOR: '' }
   882→      data.rates.forEach((r) => { map[r.providerType] = r.rate })
   883→      setRates(map)
   884→    }
   885→  }, [data])
   886→
   887→  async function save() {
   888→    setSaving(true)
   889→    try {
   890→      const payload = { rates: (Object.keys(rates) as ProviderType[]).map((pt) => ({ providerType: pt, rate: rates[pt] || '0' })) }
   891→      await apiPut('/api/admin/commission', payload)
   892→      toast.success(t('admin.commissionUpdated'))
   893→      refetch()
   894→    } catch (e: any) {
   895→      toast.error(e.message || t('admin.error'))
   896→    } finally {
   897→      setSaving(false)
   898→    }
   899→  }
   900→
   901→  if (loading) {
   902→    return (
   903→      <div>
   904→        <PageHeader title={t('admin.commissionTitle')} icon="percent" />
   905→        <LoadingCard lines={4} />
   906→      </div>
   907→    )
   908→  }
   909→  if (error || !data) return <ErrorState message={error || t('admin.error')} onRetry={refetch} />
   910→
   911→  const types: ProviderType[] = ['DOCTOR', 'HOSPITAL', 'HOTEL', 'TRANSLATOR']
   912→  const iconFor = (pt: ProviderType) => pt === 'DOCTOR' ? 'medical_services' : pt === 'HOSPITAL' ? 'local_hospital' : pt === 'HOTEL' ? 'hotel' : 'translate'
   913→
   914→  return (
   915→    <div className="animate-fade-in">
   916→      <PageHeader title={t('admin.commissionTitle')} description={t('admin.commissionDesc')} icon="percent" />
   917→      <div className="grid grid-cols-1 gap-4 lg:grid-cols-3">
   918→        <Card className="lg:col-span-2">
   919→          <CardHeader>
   920→            <CardTitle className="text-base">{t('admin.commissionTitle')}</CardTitle>
   921→            <CardDescription>{t('admin.commissionDesc')}</CardDescription>
   922→          </CardHeader>
   923→          <CardContent className="flex flex-col gap-3">
   924→            {types.map((pt) => (
   925→              <div key={pt} className="flex flex-col gap-3 rounded-[14px] border border-divider bg-surface-secondary/40 p-4 sm:flex-row sm:items-center sm:justify-between">
   926→                <div className="flex items-center gap-3">
   927→                  <div className="flex size-10 items-center justify-center rounded-[10px] bg-primary/10 text-primary">
   928→                    <Icon name={iconFor(pt)} size={22} fill />
   929→                  </div>
   930→                  <div>
   931→                    <p className="text-sm font-medium text-foreground">{t(PROVIDER_TYPE_LABEL_KEY[pt])}</p>
   932→                    <p className="text-xs text-muted-foreground">{t('admin.currentRate')}: {rates[pt] || '0'}%</p>
   933→                  </div>
   934→                </div>
   935→                <div className="flex items-center gap-2">
   936→                  <Input
   937→                    type="number"
   938→                    min="0"
   939→                    max="100"

  1578→function ReportsSection() {

  1585→    const byTypeMap: Record<string, number> = {}
  1586→    const dailyMap: Record<string, number> = {}
  1587→
  1588→    data.entries.forEach((e) => {
  1589→      const amt = parseFloat(e.amount)
  1590→      if (e.type === 'COMMISSION') {
  1591→        commission += amt
  1592→        const day = new Date(e.createdAt).toISOString().slice(0, 10)
  1593→        dailyMap[day] = (dailyMap[day] || 0) + amt
  1594→      } else if (e.type === 'REFUND_COMMISSION_REVERSAL') {
  1595→        commissionReversal += Math.abs(amt)
  1596→        const day = new Date(e.createdAt).toISOString().slice(0, 10)
  1597→        dailyMap[day] = (dailyMap[day] || 0) - Math.abs(amt)
  1598→      } else if (e.type === 'PATIENT_CHARGE') {
  1599→        patientCharge += amt
  1600→      } else if (e.type === 'REFUND_PATIENT') {
  1601→        refundPatient += Math.abs(amt)
  1602→      } else if (e.type === 'PAYOUT') {
  1603→        payout += Math.abs(amt)
  1604→      }
  1605→    })
  1606→
  1607→    const platformRevenue = commission - commissionReversal
  1608→    const dailyRevenue = Object.entries(dailyMap)
  1609→      .map(([date, amount]) => ({ date, amount }))
  1610→      .sort((a, b) => a.date.localeCompare(b.date))
  1611→      .slice(-14)
  1612→
  1613→    return { platformRevenue, commission, commissionReversal, patientCharge, refundPatient, payout, dailyRevenue }
