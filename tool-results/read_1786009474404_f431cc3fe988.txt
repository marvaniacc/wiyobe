   705→  onBooked: () => void

   793→        onOpenChange(false)
   794→        onBooked()

   796→      .catch((e: any) => {

  1001→
  1002→            <DialogFooter className="flex-row gap-2 sm:flex-row">

  1014→            </DialogFooter>
  1015→          </div>
  1016→        )}
  1017→      </DialogContent>
  1018→    </Dialog>

  1063→  }, [type, debouncedQ, city, maxPrice, sort])
  1064→
  1065→  const { data, loading, error, refetch } = useApi<{ results: Provider[]; count: number }>(url, { deps: [url, bookTick] })
  1066→
  1067→  const typeTabs: { key: 'all' | ProviderType; label: string; icon: string }[] = [

  1080→      <Card className="py-0">
